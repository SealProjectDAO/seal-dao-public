<section>
  <h2>Quick Start</h2>
  <div class="steps">
    <div class="step"><div><h4>Build from source</h4>
      <pre>git clone https://github.com/SealProjectDAO/seal-dao-master.git
cd seal-dao-master
cargo build --release -p seal-node -p seal-cli</pre>
    </div></div>
    <div class="step"><div><h4>Start a node</h4>
      <pre>./target/release/seal-node --slots 0 --rpc-port 8545</pre>
      <p>Runs forever, serves JSON-RPC on port 8545.</p>
    </div></div>
    <div class="step"><div><h4>Query via CLI</h4>
      <pre>seal-cli sql "SELECT * FROM users" --node http://localhost:8545
seal-cli sql "INSERT INTO users VALUES (3, 'charlie', 750)" --node http://localhost:8545</pre>
    </div></div>
    <div class="step"><div><h4>Query via curl</h4>
      <pre>curl -s localhost:8545 -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","method":"seal_querySql",
       "params":{"sql":"SELECT * FROM users"},"id":1}'</pre>
    </div></div>
  </div>
</section>

<section>
  <h2>RPC API</h2>
  <table>
    <tr><th>Method</th><th>Auth</th><th>Description</th></tr>

    <tr><td><code>seal_getHeight</code></td><td>No</td><td>Get current block height</td></tr>
    <tr><td><code>seal_getStateRoot</code></td><td>No</td><td>Get current state root hash</td></tr>
    <tr><td><code>seal_getBlock</code></td><td>No</td><td>Get block by height</td></tr>
    <tr><td><code>seal_deployNamespace</code></td><td>ML-DSA</td><td>Deploy a new app namespace</td></tr>
    <tr><td><code>seal_createToken</code></td><td>ML-DSA</td><td>Create a custom token</td></tr>
    <tr><td><code>seal_mintToken</code></td><td>ML-DSA</td><td>Mint tokens to an address</td></tr>
